<!-- JS program to find the largest element of an array using map() method -->
 <!DOCTYPE html>
<html>
<head>
    <title>Square of Elements using map()</title>
</head>
<body>
    <script>
        var n = parseInt(prompt("Enter the number of elements in the array:"));
        var numbers = [];

        for (var i = 0; i < n; i++) {
            var num = parseInt(prompt("Enter number " + (i + 1) + ":"));
            numbers.push(num);
        }
        numbers.map(function(num) {
            document.write("Square of element is: " (num*num));
        });
        document.write("Array:"+numbers.join(",")+"<br>")
        
    </script>
</body>
</html>
