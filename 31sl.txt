<!DOCTYPE html>
<head>
    <title>JavaScript Keyboard Events</title>
</head>
<body>
    <h2>JavaScript Keyword Events</h2>
    <input type="text"
    onkeydown="keydown()"
    onkeyup="keyup()"
    onkeypress="keypress()"
    placeholder="Press/ Type something">
    <p id="result"></p>
    <script>
        function keydown(){
            document.getElementById("result").innerHTML="A key was pressed (Keydown)";
        }
        function keyup(){
            document.getElementById("result").innerHTML="A key was pressed (Keyup)";
        }
        function keypress(){
            document.getElementById("result").innerHTML="A key was pressed (Keypress)";
        }
    </script>
</body>
